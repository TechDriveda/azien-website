CREATE DATABASE IF NOT EXISTS azian_db;
USE azian_db;

CREATE TABLE IF NOT EXISTS membership (
  membership_number INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  firstname         VARCHAR(100)  NOT NULL,
  lastname         VARCHAR(100)  NOT NULL,
  email        VARCHAR(255)  NOT NULL UNIQUE,
  phone        VARCHAR(40)   NULL,
  access       TINYINT(1)    NOT NULL DEFAULT 0,
  asset_owner  TINYINT(1)    NOT NULL DEFAULT 0,
  dealer       TINYINT(1)    NOT NULL DEFAULT 0,
  basic        TINYINT(1)    NOT NULL DEFAULT 0,
  plus         TINYINT(1)    NOT NULL DEFAULT 0,
  premium      TINYINT(1)    NOT NULL DEFAULT 0,
  created_at   TIMESTAMP     NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 AUTO_INCREMENT=100;
